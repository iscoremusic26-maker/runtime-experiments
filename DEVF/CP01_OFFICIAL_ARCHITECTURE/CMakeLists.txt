cmake_minimum_required(VERSION 3.25)
project(DEVFOfficialArchitecture LANGUAGES C CXX)
if(NOT DEFINED DUSK_SOURCE OR NOT EXISTS "${DUSK_SOURCE}/sdk/CMakeLists.txt")
    message(FATAL_ERROR "DUSK_SOURCE must be the clean exact official v2.0.3 checkout")
endif()
add_subdirectory("${DUSK_SOURCE}/sdk" sdk EXCLUDE_FROM_ALL)
add_mod(devf_official_disarmed SOURCES prototype/mod.cpp MOD_JSON prototype/mod.json)
add_executable(official_placement_lab prototype/lab.cpp prototype/mod.cpp)
target_link_libraries(official_placement_lab PRIVATE dusklight_mod_api)
target_compile_features(official_placement_lab PRIVATE cxx_std_20)
add_library(game_abi_surface OBJECT prototype/game_abi_surface.cpp)
target_link_libraries(game_abi_surface PRIVATE dusklight_game_abi_headers)
target_compile_features(game_abi_surface PRIVATE cxx_std_20)
if(MSVC)
    # Dolphin types.h gates nullptr compatibility on __cplusplus, as prior CP01 documented.
    target_compile_options(game_abi_surface PRIVATE /FIglobal.h /bigobj /utf-8 /Zc:__cplusplus)
endif()
enable_testing()
add_test(NAME official_placement_lab COMMAND official_placement_lab)
